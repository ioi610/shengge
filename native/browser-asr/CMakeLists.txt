cmake_minimum_required(VERSION 3.20)
project(shengge_browser_asr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -O3 -msimd128 -pthread")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O3 -msimd128 -pthread")
set(GGML_WEBGPU ON CACHE BOOL "" FORCE)
add_subdirectory(${WHISPER_SOURCE} whisper)
add_executable(shengge-asr bridge.cpp)
target_include_directories(shengge-asr PRIVATE ${WHISPER_SOURCE}/examples)
target_link_libraries(shengge-asr PRIVATE whisper)
target_link_options(shengge-asr PRIVATE --bind -O3 -pthread -sMODULARIZE=1 -sEXPORT_ES6=1 -sEXPORT_NAME=createShenggeAsr -sENVIRONMENT=worker -sDYNAMIC_EXECUTION=0 -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=536870912 -sMAXIMUM_MEMORY=2147483648 -sSTACK_SIZE=5242880 -sPTHREAD_POOL_SIZE=4 -sPTHREAD_POOL_SIZE_STRICT=0 -sFORCE_FILESYSTEM=1 "-sEXPORTED_RUNTIME_METHODS=['FS','HEAPU8','HEAPF32']")
set_target_properties(shengge-asr PROPERTIES SUFFIX ".mjs")
